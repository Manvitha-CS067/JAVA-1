#include<stdio.h>
#define MAX_ITEM 10
int KnapSack(int W,int Wt[],int Val[],int n);
int max(int a,int b);
int main(){
int i;
int n;
int W;
int Wt[MAX_ITEM];
int Val[MAX_ITEM];
printf("Enter the number of items: ");
scanf("%d",&n);
printf("Enter the weights of the items:\n");
for(i=1;i<=n;i++){
scanf("%d",&Wt[i]);
}

 printf("Enter the values of the items:\n");
for(i=1;i<=n;i++){
scanf("%d",&Val[i]);
}
printf("Enter the capacity of the knapsack: ");
scanf("%d",&W);
printf("The maximum value in knapsack of the capacity %d is %d\n",W,KnapSack(W,Wt,Val,n));
return 0;
}

int KnapSack(int W,int Wt[],int Val[],int n){
int i,j;
int v[MAX_ITEM +1][W+1];

for(i=0;i<=n;i++){
for(j=0;j<=W;j++){
if(i==0 ||j==0){
v[i][j]=0;
}
else if(j-Wt[i]<0){
v[i][j]=v[i-1][j];
}
else{
v[i][j]=max(v[i-1][j],Val[i]+v[i-1][j-Wt[i]]);
}
}
}
return v[n][W];}
int max(int a,int b){
return(a>b)?a:b;
}
